# Until defaults/target-packages.txt is back on master (it was dropped by
# 37ccaa1), every board carries its own list.  Once the defaults land this
# shrinks to the genuine extras and the rest comes for free.

# Base userland.  grow-rootfs needs no package of its own: it resizes with
# sfdisk and partx, both from util-linux, which @system already pulls in.
net-misc/ntp
app-admin/metalog
sys-apps/busybox
app-editors/vim
sys-process/htop
app-misc/fastfetch
dev-vcs/git
app-portage/gentoolkit

# The point of this board is the mainline media stack, and none of it can be
# checked from a serial console without v4l2-ctl.  In Gentoo that lives in
# media-libs/libv4l (the old media-tv/v4l-utils is blocked by it), behind the
# `utils` USE flag, which is on by default.  It answers the only question that
# matters on first boot:
#   v4l2-ctl --list-devices | grep -A2 snps_hdmirx
media-libs/libv4l

# Mesa for the Mali-G610: panfrost GL + panvk Vulkan.  VIDEO_CARDS is written
# into the crossdev prefix by pre-deps.sh.  Cross-emerge of mesa is not proven
# yet; dropping this line leaves a bootable image with the kernel-side GPU,
# NPU and VPU drivers still in place.
media-libs/mesa

# Desktop.  sway is wlroots, so there is no X server anywhere in here; USE
# carries -X, and see make.conf.
gui-wm/sway
gui-apps/foot
gui-apps/wl-clipboard
gui-apps/wmenu
media-video/pipewire
media-video/wireplumber

# sway is wlroots, and wlroots needs a seat manager.  Without logind that is
# seatd, and without seatd sway does not start at all.
sys-auth/seatd
media-sound/alsa-utils

# The point of the board.  Mainline RK3588 has no H.264 encoder -- the only
# encoder in the tree is JPEG (hantro_h1_jpeg_enc, rockchip_vpu2_hw_jpeg_enc)
# -- so H.264 is software or it is nothing.  x264 forces the gpl USE flag.
media-video/ffmpeg
media-libs/x264
media-video/mpv

# GStreamer for the capture pipeline: v4l2src reads the HDMI receiver,
# v4l2convert reaches RGA for colour conversion, x264enc encodes, srtsink
# ships it.  1.26 has all of these; only RK3588 stateless decode via
# v4l2codecs would need 1.28.
media-libs/gstreamer
media-libs/gst-plugins-base
media-libs/gst-plugins-good
media-libs/gst-plugins-bad
media-plugins/gst-plugins-v4l2
media-plugins/gst-plugins-x264
media-plugins/gst-plugins-srt
media-plugins/gst-plugins-jpeg
media-plugins/gst-plugins-opus
